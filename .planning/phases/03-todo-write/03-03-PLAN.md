```rust
/// Get a clone of the shared TodoManager for tool creation
pub fn get_todo_manager(&self) -> Arc<Mutex<TodoManager>> {
    self.todo_manager.clone()
}

/// Check if nag reminder should be shown
fn should_nag(&self) -> bool {
    if self.rounds_since_todo < 3 {
        return false;
    }
    // Check if there are any todos
    if let Ok(manager) = self.todo_manager.lock() {
        !manager.is_empty()
    } else {
        false
    }
}

/// Reset the todo counter (called when todo tool is used)
fn reset_todo_counter(&mut self) {
    self.rounds_since_todo = 0;
}

/// Increment the round counter (called after each agent response)
fn increment_round(&mut self) {
    self.rounds_since_todo += 1;
}
```
