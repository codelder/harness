    match block {
        TimelineBlock::UserMessage { turn_id, text } => format!("You [{}]\n{}", turn_id, text),
        TimelineBlock::AssistantMessage { turn_id, text } => {

With simplified format:
```rust
fn render_timeline_block(block: &TimelineBlock) -> String {
    match block {
        TimelineBlock::UserMessage { text, .. } => format!(">>> You\n{}", text),
        TimelineBlock::AssistantMessage { text, .. } => {
            format!("<<< Assistant\n{}", text)
        }
        TimelineBlock::Thinking { text, .. } => {
            format!("... Thinking\n{}", text)
        }
        TimelineBlock::Note {
            kind,
            message,
            ..
        } => {
            let label = match kind {
                NoteKind::Reminder => "! Reminder",
                NoteKind::Error => "!! Error",
                NoteKind::Status => "* Status",
            };
            format!("{}\n{}", label, message)
        }
        TimelineBlock::Tool(tool) => {
            let mut lines = vec![format!("[{}]", tool.name)];
            if !tool.args_preview.is_empty() {
                lines.push(format!("  {}", tool.args_preview));
            }
            if let Some(result_preview) = &tool.result_preview {
                lines.push(format!("  -> {}", result_preview));
            }
            lines.join("\n")
        }
    }
}
```

Key changes:
- User: ">>> You" instead of "You [turn_id]"
- Assistant: "<<< Assistant" instead of "Assistant [turn_id]"
- Thinking: "... Thinking" instead of "Thinking [turn_id]"
- Tool: "[name]" with indented args/result instead of "Tool [turn_id:call_id] name"
- Notes: Use symbols (!, !!, *) instead of verbose labels with turn_id
